
name: Build MAGIC IPTV APK

on:
  workflow_dispatch:
  push:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Java
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.7'

      - name: Prepare Android project
        run: |
          mkdir -p app/src/main/java/com/magic/iptv
          mkdir -p app/src/main/res/values
          mkdir -p app/src/main/res/drawable
          mkdir -p app/src/main/assets

          cp app_build.gradle app/build.gradle
          cp debug.keystore app/debug.keystore
          cp AndroidManifest.xml app/src/main/
          cp MainActivity.java app/src/main/java/com/magic/iptv/
          cp PlaybackService.java app/src/main/java/com/magic/iptv/
          cp strings.xml app/src/main/res/values/
          cp ic_launcher.xml banner.xml ic_stat.xml app/src/main/res/drawable/

          cp index.html app/src/main/assets/

      - name: Build APK
        run: gradle assembleDebug --no-daemon

      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: MAGIC-APK
          path: app/build/outputs/apk/debug/*.apk
          if-no-files-found: error
